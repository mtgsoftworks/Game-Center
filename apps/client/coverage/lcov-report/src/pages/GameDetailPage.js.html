  const params = useParams();
  const gameId = params.gameId || params.id;
  const navigate = useNavigate();
  const { t } = useTranslation();

  // Firestore'dan oyun verisini yükle, yoksa fallback sampleGameData kullan
  useEffect(() => {
    const fetchGame = async () => {
      setLoading(true);
      try {
        const snap = await getDoc(doc(db, 'games', gameId));
        if (snap.exists()) {
          setGame({ id: snap.id, ...snap.data() });
        } else if (sampleGameData[gameId]) {
          setGame(sampleGameData[gameId]);
          setError(null);
        } else {
          setError(t('gameDetailPage.notFound', 'Oyun bulunamadı!'));
        }

        console.error(err);

        setLoading(false);
      }

    fetchGame();

      </Container>
